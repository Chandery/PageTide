capacity.py 源码：
"""Per-batch capacity of the category_review wire: base commit 9fdae25 vs this work unit.

Recipe: English public text at the cap (title 80, abstract 160 characters); category names of N Chinese
characters (default 6, e.g. 研究方向名称1); all categories top-level; all papers unclassified; 8000-byte cap.
Prints the largest number of papers (max 12) one batch can carry for N_CATEGORIES categories, and the largest
category directory that fits with zero papers."""
import importlib.util, sys
sys.path.insert(0, '.')
from app import category_review as new

spec = importlib.util.spec_from_file_location('base_cr', '/tmp/claude-501/base_cr.py')
base = importlib.util.module_from_spec(spec); spec.loader.exec_module(base)

def papers(count, abstract='a' * 160):
    return [{'key': 'k%d' % i, 'title': 'T' * 80, 'abstract': abstract, 'categories': []} for i in range(count)]

def directory(count, name_chars):
    return [{'id': 'c%d' % i, 'name': ('研究方向名称%d' % i) if name_chars == 6 else ('名' * name_chars + str(i)), 'parent_id': None} for i in range(count)]

def per_batch(module, cats, name_chars, abstract):
    for count in range(12, 0, -1):
        try:
            module.prepare(directory(cats, name_chars), papers(count, abstract)); return count
        except ValueError:
            continue
    return 0

def max_directory(module, name_chars):
    best = 0
    for cats in range(1, 61):
        try:
            module.prepare(directory(cats, name_chars), []); best = cats
        except ValueError:
            break
    return best

print('system prompt bytes: base', len(base.SYSTEM.encode()), 'new', len(new.SYSTEM.encode()), 'delta', len(new.SYSTEM.encode()) - len(base.SYSTEM.encode()))
print('papers per batch (English text, 6-char names)        categories: base -> new')
for cats in (0, 10, 20, 30, 40, 50, 60):
    print('  %2d categories: %2d -> %2d' % (cats, per_batch(base, cats, 6, 'a' * 160), per_batch(new, cats, 6, 'a' * 160)))
print('papers per batch (Chinese abstracts of 160 characters, 6-char names)')
for cats in (0, 10, 20, 40):
    print('  %2d categories: %2d -> %2d' % (cats, per_batch(base, cats, 6, '中' * 160), per_batch(new, cats, 6, '中' * 160)))
print('largest directory that fits with zero papers, by category-name length (characters): base -> new')
for chars in (6, 12, 20, 30):
    print('  %2d characters: %2d -> %2d' % (chars, max_directory(base, chars), max_directory(new, chars)))
print('60 categories x 12 papers with empty public text:')
ds = [{'id': 'c%d' % i, 'name': str(i), 'parent_id': None} for i in range(60)]
ps = [{'key': 'k%d' % i, 'title': '', 'abstract': '', 'categories': []} for i in range(12)]
for label, module in (('base', base), ('new ', new)):
    try:
        print('  ', label, 'estimate', module.message_estimate(module.prepare(ds, ps)['messages']), '/ cap', module.INPUT_CAP)
    except ValueError as error:
        print('  ', label, 'refused:', error)

## 7. 真实浏览器验证（手工，非命令；合成数据，只桩住 review-categories 这一个模型任务，其余请求含 apply-ai-review 走真实服务端）

- 项目标签“世界模型研究”：左侧目录 4 个类别（“固态电池”只在其他项目，被收起），预览范围句与之一致。
- 三段顺序：先归类（按去向分组，新类别显示“父类 › 名称”与理由，“其他”标“依据不足，暂放”）→ 调整现有分类（合并、删除标签及其改归的论文可展开查看）→ 重新归类已分类论文（全部项目标签的合成数据）。
- 勾选联动：取消“删除标签”与“其他”组 → 汇总“已选 4 / 6 项 · 应用后仍有 1 篇未分类”，按钮“应用所选（4 项）”；父类删除依赖子类删除：取消子类 → 父类自动取消，勾父类 → 子类自动勾上。
- 应用后服务端实际：只应用了勾选的 4 项（合并 World Models→世界模型；wm-1/2/3 归类；视频世界模型新建），未勾选的删除/“其他”未动，其他项目的“固态电池”未动；提示语“…仍有 1 篇未分类 · 可以撤销”。一次撤销整体还原。
- 全选应用（全部项目）后 unclassified=[]，“其他”只收了 AI 判断不了的那一篇。
- 报错：预览变陈旧后点应用，红字报错在粘性底栏内、滚到顶部仍在可视区（修复前会被挡在列表下面）；英文界面下报错句、范围句、汇总句均已翻译，PDHi18n.misses() 为空（剩下的中文都是用户数据与手写种子里的中文理由）。
